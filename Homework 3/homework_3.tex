\section*{Problem 1}
\begin{enumerate}[label=(\arabic*)]
    \item    
    \[
    ij = k, \quad ij = -ji \implies -ji = k \implies ji = -k
    \]
    Hence, \( ij \neq ji \), so multiplication is not commutative and \( \mathbb{H} \) is a non-commutative ring.
\end{enumerate}
