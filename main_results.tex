\section{Groups}
\subsection{Conjugation, centralizer, center, normalizer}
For $g\in G$, the conjugation map $c_g(x)=gxg^{-1}$ is an automorphism of $G$ (an \emph{inner} automorphism).
For a subset $S\subseteq G$, the centralizer is
\[
C_G(S)=\{\,g\in G:\ gs=sg\ \text{ for all }s\in S\,\};\quad C_G(g):=C_G(\{g\}).
\]
The center is 
\[ 
    Z(G)=C_G(G)=\{g\in G:\ gx=xg\ \forall x\in G\}
\]
(always a characteristic, hence normal, subgroup).
For a subgroup $H\le G$, the normalizer is
\[
N_G(H)=\{\,g\in G:\ gHg^{-1}=H\,\},
\]
the largest subgroup of $G$ in which $H$ is normal (so $H\trianglelefteq N_G(H)$). One always has $C_G(H)\le N_G(H)$.
The number of conjugates of $H$ in $G$ equals $[G:N_G(H)]$.

\subsection{Normal subgroups and quotients}
A subgroup $N\le G$ is normal (write $N\trianglelefteq G$) if $gNg^{-1}=N$ for all $g\in G$.
Equivalently, every left coset equals the corresponding right coset: $gN=Ng$.
If $N\trianglelefteq G$, the set of cosets $G/N=\{gN:g\in G\}$ is a group with $(gN)(hN)=(gh)N$.

\subsection{Cosets and index}
For $H\le G$ and $g\in G$, the left (resp.\ right) coset is
$gH=\{gh:h\in H\}$ (resp.\ $Hg=\{hg:h\in H\}$).
Left cosets are either disjoint or equal and they partition $G$.
Each left coset $gH$ has $|H|$ elements via the bijection $H\to gH$, $h\mapsto gh$.
The \emph{index} of $H$ in $G$ is the number of left cosets, denoted $[G\!:\!H]$.

\subsection{Lagrange's Theorem}
If $G$ is finite and $H\le G$, then
\[
|G| \;=\; [G\!:\!H]\cdot |H|.
\]
\emph{Proof sketch.} $G$ is the disjoint union of $[G\!:\!H]$ left cosets, each of cardinality $|H|$.

\paragraph{Consequences}
\begin{itemize}[leftmargin=1.2em]
  \item $|H|\mid |G|$ for every $H\le G$.
  \item For $g\in G$, the order $|g|$ divides $|G|$; hence $g^{\,|G|}=e$.
  \item Index multiplicativity: for $K\le H\le G$ (with $G$ finite),
        \([G\!:\!K]=[G\!:\!H]\,[H\!:\!K]\).
  \item If $[G\!:\!H]=2$, then $H\trianglelefteq G$ (there are only two cosets, so $gHg^{-1}$ must be $H$).
\end{itemize}

\paragraph{Groups of prime order are cyclic.}
If $|G|=p$ (prime) and $g\in G$, $g\neq e$, then $|\langle g\rangle|=|g|\mid p$ by Lagrange,
so $|g|=p$ and $\langle g\rangle=G$. Hence $G$ is cyclic and $G\cong \mathbb{Z}/p\mathbb{Z}$.

\subsection{Isomorphism theorems}
\begin{enumerate}[label=(\arabic*)]
\item \textbf{First isomorphism theorem.} For $\varphi:G\to H$ a homomorphism,
\[
G/\ker\varphi \ \cong\ \mathrm{im}\,\varphi.
\]
\item \textbf{Second isomorphism theorem.} For $A\le G$ and $N\trianglelefteq G$,
\[
A\cap N\trianglelefteq A,\qquad AN\le G,\qquad AN/N\ \cong\ A/(A\cap N).
\]
\item \textbf{Third isomorphism theorem.} If $K\le H\trianglelefteq G$, then $H/K\trianglelefteq G/K$ and
\[
(G/K)/(H/K)\ \cong\ G/H.
\]
\item \textbf{Correspondence (lattice) theorem.} For $N\trianglelefteq G$, the map
\[
\{\,\text{subgroups }H\mid N\le H\le G\,\}\ \longleftrightarrow\ \{\,\text{subgroups of }G/N\,\},\quad
H\mapsto H/N
\]
is a bijection preserving inclusion and index; it carries normal subgroups $H\trianglelefteq G$ with $N\le H$ to normal subgroups of $G/N$.
\end{enumerate}

\section{Group actions}

\subsection{Actions on cosets and by conjugation}
\begin{itemize}[leftmargin=1.2em]
  \item For $H\leq G$, the action $G\curvearrowright G/H$ by left multiplication is transitive, with stabilizer of $aH$ equal to $aHa^{-1}$. 
  \item The action $G\curvearrowright G$ by conjugation, $g\cdot x=gxg^{-1}$, partitions $G$ into conjugacy classes. 
        The stabilizer of $x$ is its centralizer $C_G(x)$, and $|\mathrm{Orb}(x)|=[G:C_G(x)]$.
\end{itemize}

\subsection{Cayley's Theorem}
Every group $G$ is isomorphic to a subgroup $H$ of the symmetric group $S_{|G|}$:
\[
G \cong H \leq S_{|G|}.
\]

\subsection{Orbit--stabilizer}
For any action $G\curvearrowright X$ and any $x\in X$ there is a canonical bijection
\[
G/G_x\ \xrightarrow{\ \sim\ }\ \mathrm{Orb}(x),\qquad gG_x\longmapsto g\cdot x.
\]
In particular, if $G$ is finite then
\[
|\mathrm{Orb}(x)|=[G:G_x]\quad\text{and}\quad |G|=|\mathrm{Orb}(x)|\cdot |G_x|.
\]

\subsection{Class Equation}
For a finite group $G$, let $Z(G)$ be the center of $G$. Then
\[
|G| \;=\; |Z(G)| \;+\; \sum_{i=1}^r \,[G : C_G(g_i)],
\]
where $g_1, \dots, g_r$ are representatives of the conjugacy classes not contained in $Z(G)$.

\subsection{Conjugacy in $S_n$}
Two permutations in $S_n$ are conjugate if and only if they have the same cycle type
(i.e.\ the same cycle lengths with the same multiplicities).
\[
\sigma, \tau \in S_n \quad\text{are conjugate} \;\;\Longleftrightarrow\;\;
\text{cycle-type}(\sigma) = \text{cycle-type}(\tau).
\]
Hence, the conjugacy classes of $S_n$ correspond bijectively to partitions of $n$.

\subsection{Automorphisms of cyclic groups}
For $n\ge1$, the automorphism group of the cyclic group $C_n\cong \mathbb Z/n\mathbb Z$ is
\[
\mathrm{Aut}(C_n)\ \cong\ (\mathbb Z/n\mathbb Z)^\times,
\]
hence $|\mathrm{Aut}(C_n)|=\varphi(n)$. Here $\varphi(n)$ is \emph{Euler's totient function}:
\[
\varphi(n)=\big|(\mathbb Z/n\mathbb Z)^\times\big|
= |\{\,1\le k\le n:\gcd(k,n)=1\,\}|.
\]
Equivalently, if $n=\prod p_i^{\,e_i}$ then
\[
\varphi(n)=n\!\prod_{i}\!\Big(1-\frac1{p_i}\Big)
=\prod_{i}\!\big(p_i^{e_i}-p_i^{e_i-1}\big).
\]


\section{Sylows Theorem}
Let $|G|=p^{a}m$ with $p\nmid m$ and $a\ge1$. A subgroup of order $p^{a}$ is a \emph{Sylow $p$-subgroup}.  
Write $\mathrm{Syl}_p(G)$ for the set of Sylow $p$-subgroups and $n_p:=|\mathrm{Syl}_p(G)|$.

\subsection{Statements}
\begin{itemize}[leftmargin=1.2em]
  \item $\mathrm{Syl}_p(G) \neq \varnothing $.
  \item Every $p$-subgroup of $G$ is contained in some conjugate of a Sylow $p$-subgroup. In particular, any two Sylow p-subgroups of $G$ are conjugate in $G$.
  \item $G$ acts on $\mathrm{Syl}_p(G)$ by $g\cdot P = gPg^{-1}$.  
  For $P\in\mathrm{Syl}_p(G)$, the stabilizer is $N_G(P)$, hence
  \[
    n_p = [G : N_G(P)].
    \]
  \item $n_p \equiv 1 \pmod p$ and $n_p \mid m$.
  \item If $n_p=1$, then the unique Sylow $p$-subgroup $P$ is normal \emph{and characteristic} in $G$
        (automorphisms permute Sylow $p$-subgroups; uniqueness forces $P$ to be fixed).
\end{itemize}

\section{Rings}
\subsection{Chinese Remainder Theorem}
Let $R$ be a commutative ring with $1$ and let $A_1,\dots,A_k\unlhd R$ be ideals.
If the $A_i$ are pairwise comaximal ($A_i+A_j=R$ for $i\neq j$), then the natural map
\[
\phi:R \longrightarrow \prod_{i=1}^k R/A_i,\qquad r\mapsto (r+A_i)_i
\]
is surjective with kernel $\bigcap_{i=1}^k A_i=\prod_{i=1}^k A_i$. Hence
\[
R\big/\!\big(\textstyle\prod_{i=1}^k A_i\big)\ \cong\ \prod_{i=1}^k R/A_i .
\]

\section{Integral -, Unique Factorization - \& Principle Ideal Domains}
\subsection{Gauss's Lemma}
Let $R$ be a UFD with field of fractions $F$, and let $p(x)\in R[x]$. 
If $p$ is reducible in $F[x]$, then $p$ is reducible in $R[x]$.
(Equivalently: if $p$ is irreducible in $R[x]$, then it is irreducible in $F[x]$.)

\subsection{Eisenstein's Criterion}
Let $R$ be an integral domain and $P\subset R$ a prime ideal. Consider
\[
f(x)=x^n+a_{n-1}x^{n-1}+\cdots+a_1x+a_0\in R[x].
\]
If $a_{n-1},\dots,a_0\in P$ and $a_0\notin P^2$, then $f(x)$ is irreducible in $R[x]$.

\paragraph{Corollary ($\mathbb{Z}[x]$).}
Let $f(x)=x^n+a_{n-1}x^{n-1}+\cdots+a_0\in\mathbb{Z}[x]$. 
If there exists a prime $p$ such that $p\mid a_i$ for $i=0,\dots,n-1$ and $p^2\nmid a_0$, 
then $f$ is irreducible in $\mathbb{Z}[x]$ (hence in $\mathbb{Q}[x]$ by Gauss).
\section{Field Extensions}
\subsection{Simple extensions and minimal polynomials}
If $\alpha$ is algebraic over a field $F$, there is a unique monic irreducible
$m_\alpha(x)\in F[x]$ with $m_\alpha(\alpha)=0$ (the \emph{minimal polynomial} of $\alpha$).

\subsection{Construction via quotients}
If $p(x)\in F[x]$ is irreducible and $K/F$ contains a root $\alpha$ of $p$, then
\[
F(\alpha)\ \cong\ F[x]/(p(x)).
\]
In particular, for any irreducible $p$, the quotient $F[x]/(p)$ is a field of degree $\deg p$,
and the class $\bar x:=x+(p)$ is a root of $p$.

\subsection{Degree and basis of a simple extension}
Let $\alpha$ be algebraic over $F$ with $m_\alpha$ of degree $n$. Then
\[
[F(\alpha):F]=n,\qquad \{1,\alpha,\alpha^2,\dots,\alpha^{n-1}\}\ \text{is an $F$-basis of }F(\alpha).
\]

\subsection{Tower law}
For fields $F\subseteq K\subseteq L$,
\[
[L:F]=[L:K]\,[K:F].
\]

\subsection{Finite $\Rightarrow$ algebraic}
If $K/F$ is a finite extension, then every $\alpha\in K$ is algebraic over $F$ (hence $K/F$ is algebraic).
